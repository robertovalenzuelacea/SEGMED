  <script>
    // 1. Auto-completar desde parámetros URL
    const params = new URLSearchParams(window.location.search);
    if (params.get('lead_id')) document.getElementById('lead_id').value = params.get('lead_id');
    if (params.get('nombre')) document.getElementById('nombre').value = decodeURIComponent(params.get('nombre'));
    if (params.get('email')) document.getElementById('email').value = decodeURIComponent(params.get('email'));
    if (params.get('empresa')) document.getElementById('empresa').value = decodeURIComponent(params.get('empresa'));

    // 2. Validar fecha actual
    const today = new Date().toISOString().split('T')[0];
    document.getElementById('fecha').min = today;
    if (!document.getElementById('fecha').value) document.getElementById('fecha').value = today;

    // 3. Envío al Backend de Render
    document.getElementById('agendaForm').addEventListener('submit', async (e) => {
      e.preventDefault();
      const btn = document.getElementById('btnSubmit');
      btn.disabled = true;
      btn.innerText = '⏳ Registrando visita...';

      const payload = {
        lead_id: document.getElementById('lead_id').value || 0,
        empresa: document.getElementById('empresa').value,
        nombre: document.getElementById('nombre').value,
        telefono: document.getElementById('telefono').value,
        email: document.getElementById('email').value,
        direccion: document.getElementById('direccion').value,
        fecha: document.getElementById('fecha').value,
        hora: document.getElementById('hora').value
      };

      try {
        await fetch('/api/agendar-visita', {
          method: 'POST',
          headers: { 'Content-Type': 'application/json' },
          body: JSON.stringify(payload)
        });
      } catch (err) {
        console.warn('Registro enviado:', err);
      }

      document.getElementById('formContainer').style.display = 'none';
      document.getElementById('successView').style.display = 'flex';
    });
  </script>
